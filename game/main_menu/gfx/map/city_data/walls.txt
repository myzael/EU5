template large_city_walls
{
	default_variables =
	{
		wall_mesh_list  =  fallback_wall
		tower_mesh_list = fallback_wall_tower
		gate_mesh_list  = fallback_wall_gate
	}
	
	code [[
		wall = {
			name = large_city_walls
			order = 0
			tower_separation = 7.0
			area = 120.0
			percentage_built = 1.0
			#limit {}
			contributors = 
			{
				has_building_with_graphical_tag_and_at_least_one_level:large_city_walls = 1.0
			}
			#required_contributors = {}
			wall_mesh_list  = @wall_mesh_list
			tower_mesh_list = @tower_mesh_list
			gate_mesh_list  = @gate_mesh_list
			#only_in_location = london
			#location_filters = { rural_settlement = no }
			#dlc = the_great_wall_dlc
			#link_tooltop = not_implemented_yet
		}
	]]
}
