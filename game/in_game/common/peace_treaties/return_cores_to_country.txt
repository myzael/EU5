return_cores_to_country = {

    category = cores
    are_targets_exclusive = yes
    blocks_full_annexation = yes

    cost = {
        add = {
            desc = "DIPLOREASON_BASE"
            value = "scope:target.location_peace_cost(scope:target.owner|scope:winner)"
            multiply = 0.5
        }
    }

    base_antagonism = {
        add = {
            desc = "ANTAGONISM_BASE"
            value = 2
        }

        if = {
            limit = { 
                scope:target.controller != scope:target.owner
                country_exists = scope:target.controller
                scope:target.controller = {
                    any_current_war = { this = scope:war }
                }
                scope:target.dominant_culture = scope:target.controller.culture
            }
            subtract = {
                desc = "SAME_CULTURE_LOCATION"
                value = 1.25
            }
        }

        add = {
            value = scope:target.development
            multiply = 0.01
            desc = "[development|e]"
        }
    }
    antagonism_type = antagonism_return_cores_to_country

    potential = {
        scope:loser = {
            any_owned_location = {
                any_core_in_location = {
                    OR = {
                        this = scope:winner
                        AND = {
                            any_current_war = { this = scope:war }
                            is_fighting_war_together_with = scope:winner
                        }
                        is_subject_or_below_of = scope:winner
                    }
                }
            }
        }
    }

    select_trigger = {
        looking_for_a = location
        source = recipient
        target_flag = target
        only_color_selectable = yes

        visible = {
            owner = scope:recipient

            any_core_in_location = {
                OR = {
                    this = scope:actor
                    AND = {
                        exists = scope:war
                        any_current_war = { this = scope:war }
                        is_fighting_war_together_with = scope:actor
                    }
                    is_subject_or_below_of = scope:actor
                }
            }
        }

        enabled = {
            controller = {
                this != scope:recipient
                
                OR = {
                    this = scope:actor
                    AND = {
                        exists = scope:war
                        any_current_war = { this = scope:war }
                        is_fighting_war_together_with = scope:actor
                    }
                    is_subject_or_below_of = scope:actor
                }
            }
        }
    }

    allow = {
        scope:target = {
            NOT = { is_capital = yes }

            any_core_in_location = {
                OR = {
                    this = scope:winner
                    AND = {                        
                        exists = scope:war
                        any_current_war = { this = scope:war }
                        is_fighting_war_together_with = scope:winner
                    }
                    is_subject_or_below_of = scope:winner
                }
            }
        }
    }
    
    effect = {
        scope:target = {

            if = {
                limit = { 
                    always = yes
                    scope:target.controller = { is_subject = yes }
                }

                scope:target.controller = {
                    add_liberty_desire = {
                        value = -1
                        multiply = {
                            value = scope:target.population
                            multiply = 0.05
                        }
                    }
                }
            }

            if = {
                limit = {
                    any_core_in_location = { 
                        this = scope:target.controller
                    }
                }
                change_location_owner = scope:target.controller
            }
            if = {
                limit = {
                    scope:winner = {
                        is_leader_of_international_organization = international_organization:hre
                    }
                    scope:target = {
                        is_owned_by_international_organization = international_organization:hre
                    }
                }

                scope:winner = {
                    change_imperial_authority = { value = 0.5 }
                }
            }

            if = {
                limit = {
                    #is_core_of = scope:loser

                    NOT = { dominant_culture = scope:loser.culture }
                }

                remove_core = scope:loser
            }


        }
    }

    ai_desire = {
        add = {
            desc = "DIPLOREASON_BASE_RELUCTANCE"
            value = -5
        }

        if = {
            limit = {
                scope:target = {
                    any_core_in_location = {
                        this = scope:winner
                    }
                }
            }
            add = {
                desc = "DIPLOREASON_CORE"
                value = 20
            }
        }
        else_if = {
            limit = {
                scope:target = {
                    any_core_in_location = {
                        this = {
                            is_subject_or_below_of = scope:winner
                            is_disloyal_subject = no
                        }
                    }
                }
            }
            add = {
                desc = "DIPLOREASON_CORE_SUBJECT"
                value = 10
            }
        }
        else = {
            add = {
                desc = "DIPLOREASON_CORE_ALLY"
                value = 5
            }
        }

        add = {
            desc = "DIPLOREASON_BASE_LOCATION_VALUE"
            value = scope:target.location_tax_base
            multiply = 0.1
        }

        if = {
            limit = {
                scope:winner = {
                    is_leader_of_international_organization = international_organization:hre
                }
                scope:target = {
                    is_owned_by_international_organization = international_organization:hre
                }
            }
            add = {
                desc = "DIPLOREASON_EMPEROR_HRE_TERRITORY"
                value = 25
            }
        }

    }

}
